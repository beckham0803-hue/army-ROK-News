        async function testNaverApiConnection() {
            const apiType = document.getElementById('apiTypeSelect').value;
            const clientId = document.getElementById('naverClientId').value.trim();
            const clientSecret = document.getElementById('naverClientSecret').value.trim();
            const proxy = document.getElementById('corsProxySelect').value;
            const diagBox = document.getElementById('apiDiagnosticBox');

            diagBox.classList.remove('hidden', 'bg-red-50', 'bg-emerald-50', 'bg-amber-50', 'text-red-800', 'text-emerald-800', 'text-amber-800', 'border-red-300', 'border-emerald-300', 'border-amber-300');

            diagBox.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-1"></i> 네이버 API 서버 통신 테스트 중...`;

            if (!clientId || !clientSecret) {
                diagBox.classList.add('bg-amber-50', 'text-amber-800', 'border-amber-300');

                return;
            }

            try {
                const { targetUrl, headers } = getApiEndpointAndHeaders(apiType, clientId, clientSecret, '육군', 1, 'date');
                let fetchUrl = targetUrl;
                if (proxy && proxy !== 'direct') {
                    fetchUrl = proxy + encodeURIComponent(targetUrl);
                }

                const response = await fetch(fetchUrl, {
                    method: 'GET',
                    headers: headers
                });

                if (response.ok) {
                    const data = await response.json();
                    diagBox.classList.add('bg-emerald-50', 'text-emerald-800', 'border-emerald-300');
                    diagBox.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-600"></i> <strong>연동 성공! (HTTP 200)</strong><br>[${apiType === 'ncp' ? 'NAVER API HUB' : '네이버 오픈 API'}] 뉴스 데이터를 성공적으로 수신했습니다.`;

                    diagBox.classList.add('bg-red-50', 'text-red-800', 'border-red-300');
                    if (response.status === 401) {
                        diagBox.innerHTML = `<i class="fa-solid fa-circle-xmark"></i> <strong>HTTP 401 (인증 실패)</strong><br>Client ID 또는 Key가 일치하지 않습니다.`;

                        diagBox.innerHTML = `<i class="fa-solid fa-circle-xmark"></i> <strong>HTTP 403 (권한 없음)</strong><br>해당 API 서비스 권한이 설정되어 있는지 확인하세요.`;
                    } else {

        async function fetchLiveNews() {
            const queryInput = document.getElementById('searchInput').value.trim();
            const query = queryInput || CONFIG.activeKeyword;
            CONFIG.activeKeyword = query;
            document.getElementById('kpiActiveTopic').textContent = query;

            const refreshIcon = document.getElementById('refreshIcon');
            if (refreshIcon) refreshIcon.classList.add('fa-spin');

            const display = document.getElementById('displaySelect').value;
            const sort = document.getElementById('sortSelect').value;

            let fetchedArticles = [];

            if (CONFIG.naverClientId && CONFIG.naverClientSecret) {
                try {
                    const { targetUrl, headers } = getApiEndpointAndHeaders(CONFIG.apiType, CONFIG.naverClientId, CONFIG.naverClientSecret, query, display, sort);
                    let finalFetchUrl = targetUrl;

                    if (CONFIG.corsProxy && CONFIG.corsProxy !== 'direct') {
                        finalFetchUrl = CONFIG.corsProxy + encodeURIComponent(targetUrl);
                    }

                    const response = await fetch(finalFetchUrl, {
                        headers: headers
                    });

                    if (response.ok) {
                        const data = await response.json();
                        if (data.items && data.items.length > 0) {
                            fetchedArticles = data.items.map((item, index) => parseNaverItem(item, index, query));
                        }
                    } else {
                        showToast(`API 오류 (HTTP ${response.status}) - 샘플 데이터로 전환`);
                    }
                } catch (err) {
                    showToast("CORS 차단으로 인해 실시간 샘플 데이터로 전환되었습니다.");
                }
            }

            if (fetchedArticles.length === 0) {
                fetchedArticles = generateFilteredMockNews(query);
            }

            newsDatabase = fetchedArticles;
            renderNewsStream();
            updateSentimentChart();

            setTimeout(() => {
                if (refreshIcon) refreshIcon.classList.remove('fa-spin');
            }, 400);
        }

        function updateSentimentChart() {
            if (!newsDatabase || newsDatabase.length === 0) {
                document.getElementById('cntPositive').textContent = '0';
                document.getElementById('cntNeutral').textContent = '0';
                document.getElementById('cntAlert').textContent = '0';
                document.getElementById('barPositive').style.width = '0%';
                document.getElementById('barNeutral').style.width = '0%';
                document.getElementById('barAlert').style.width = '0%';
                document.getElementById('kpiPositivePercent').innerHTML = `0<span class="text-xs text-slate-400 font-normal ml-1">%</span>`;
                return;
            }

            const total = newsDatabase.length;
            const posCount = newsDatabase.filter(x => x.sentiment === 'positive').length;
            const neuCount = newsDatabase.filter(x => x.sentiment === 'neutral').length;
            const alertCount = newsDatabase.filter(x => x.sentiment === 'alert').length;

            const posPct = Math.round((posCount / total) * 100);
            const neuPct = Math.round((neuCount / total) * 100);
            const alertPct = Math.max(0, 100 - posPct - neuPct);

            document.getElementById('cntPositive').textContent = posCount;
            document.getElementById('cntNeutral').textContent = neuCount;
            document.getElementById('cntAlert').textContent = alertCount;

            document.getElementById('barPositive').style.width = `${posPct}%`;
            document.getElementById('barNeutral').style.width = `${neuPct}%`;
            document.getElementById('barAlert').style.width = `${alertPct}%`;

            document.getElementById('kpiPositivePercent').innerHTML = `${posPct}<span class="text-xs text-slate-400 font-normal ml-1">%</span>`;
        }

        function parseNaverItem(item, idx, query) {
            const cleanTitle = item.title.replace(/<[^>]*>?/gm, '').replace(/&quot;/g, '"').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>');
            const cleanDesc = item.description.replace(/<[^>]*>?/gm, '').replace(/&quot;/g, '"').replace(/&amp;/g, '&').replace(/&lt;/g, '<').replace(/&gt;/g, '>');
            
            let sentiment = "neutral";
            if (/도발|발사|위협|사고|경계|북한|비상|유출|논란|위기/i.test(cleanTitle)) {
                sentiment = "alert";
            } else if (/혁신|성공|수출|격려|대승|압도|강화|도입|우수/i.test(cleanTitle)) {
                sentiment = "positive";
            }

            let mediaName = "주요 언론사";
            const link = item.originallink || item.link || '';
            if (link.includes('yna.co.kr')) mediaName = "연합뉴스";
            else if (link.includes('kbs.co.kr')) mediaName = "KBS";
            else if (link.includes('mbc.co.kr')) mediaName = "MBC";
            else if (link.includes('sbs.co.kr')) mediaName = "SBS";
            else if (link.includes('chosun.com')) mediaName = "조선일보";
            else if (link.includes('donga.com')) mediaName = "동아일보";
            else if (link.includes('joongang.co.kr')) mediaName = "중앙일보";
            else if (link.includes('ytn.co.kr')) mediaName = "YTN";
            else if (link.includes('dema.mil.kr')) mediaName = "국방일보";

            return {
                id: 'n_' + Date.now() + '_' + idx,
                title: cleanTitle,
                description: cleanDesc,
                pubDate: item.pubDate || new Date().toUTCString(),
                originallink: link,
                media: mediaName,
                keyword: query,
                sentiment: sentiment
            };
        }

        function selectKeywordChip(kw) {
            CONFIG.activeKeyword = kw;

        function deleteKeywordChip(event, index) {
            event.stopPropagation();
            const deleted = keywordsList[index];
            const deletedName = typeof deleted === 'string' ? deleted : deleted.name;

            keywordsList.splice(index, 1);
            localStorage.setItem('ROK_KEYWORDS_LIST', JSON.stringify(keywordsList));

            showToast(`'${deletedName}' 주제가 목록에서 제외되었습니다.`);

            if (CONFIG.activeKeyword === deletedName) {
                if (keywordsList.length > 0) {
                    const nextKw = typeof keywordsList[0] === 'string' ? keywordsList[0] : keywordsList[0].name;
                    selectKeywordChip(nextKw);
                } else {
                    CONFIG.activeKeyword = '육군';
                    renderKeywordChips();
                    fetchLiveNews();
                }
            } else {
                renderKeywordChips();
            }
        }

        function addCustomKeywordPrompt() {
            const val = prompt("추가할 육군 모니터링 주제 키워드를 입력하세요:");
            if (val && val.trim()) {
                const trimmed = val.trim();
                const exists = keywordsList.some(k => (typeof k === 'string' ? k : k.name) === trimmed);
                if (!exists) {
                    keywordsList.push({ name: trimmed, icon: "fa-crosshairs" });
                    localStorage.setItem('ROK_KEYWORDS_LIST', JSON.stringify(keywordsList));
                    showToast(`'${trimmed}' 주제가 추가되었습니다.`);
                }
                selectKeywordChip(trimmed);
            }
        }

        function formatDateString(str) {
            try {
                const d = new Date(str);
                if (isNaN(d.getTime())) return "오늘 보도";
                return `${d.getMonth()+1}/${d.getDate()} ${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;
            } catch(e) {
                return "오늘 보도";
            }
        }
