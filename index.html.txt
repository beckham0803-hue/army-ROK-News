        // Manual refresh button
        function manualRefresh() {
            loadNewsForKeyword(currentKeyword, selectedArchiveDate);
            countdownSeconds = timerIntervalMinutes * 60;
            showToast("실시간 뉴스를 즉시 새로고침했습니다.", "success");

        // Change Sort Order
        function changeSort(sortType) {
            currentSort = sortType;
            const latestBtn = document.getElementById('sortLatestBtn');
            const relBtn = document.getElementById('sortRelBtn');

            if (sortType === 'latest') {
                latestBtn.className = "px-2.5 py-1 rounded-md bg-army-700 text-white font-medium transition-all";
                relBtn.className = "px-2.5 py-1 rounded-md text-slate-400 hover:text-slate-200 transition-all";
            } else {
                relBtn.className = "px-2.5 py-1 rounded-md bg-army-700 text-white font-medium transition-all";
                latestBtn.className = "px-2.5 py-1 rounded-md text-slate-400 hover:text-slate-200 transition-all";
            }

            sortArticlesData();
            renderNewsFeed();
        }

        function sortArticlesData() {
            if (currentSort === 'latest') {
                newsData.sort((a, b) => b.timestamp - a.timestamp);
            } else {
                // Relevant sort (sample logic based on keyword priority)
                newsData.sort((a, b) => a.title.length - b.title.length);
            }
        }

        // Timer Interval Switcher (Requirement 3)
        function setTimerInterval(minutes) {
            timerIntervalMinutes = minutes;
            countdownSeconds = minutes * 60;

            // Highlight active button
            [5, 30, 60].forEach(m => {
                const btn = document.getElementById(`timer-${m}`);
                if (m === minutes) {
                    btn.className = "timer-btn px-3 py-2 text-xs rounded-lg font-medium border transition-all flex flex-col items-center justify-center bg-army-700 text-white border-gold-500 shadow";
                } else {
                    btn.className = "timer-btn px-3 py-2 text-xs rounded-lg font-medium border border-army-800 bg-slate-900 text-slate-300 hover:border-army-600 transition-all flex flex-col items-center justify-center";
                }
            });

            updateCountdownDisplay();
            showToast(`자동 갱신 주기가 ${minutes}분으로 변경되었습니다.`, "info");
        }

        // Countdown Timer Engine
        function startCountdownTimer() {
            if (timerId) clearInterval(timerId);

            timerId = setInterval(() => {
                countdownSeconds--;
                if (countdownSeconds <= 0) {
                    // Time reached - auto refresh
                    manualRefresh();
                    showToast(`[타이머 작동] '${currentKeyword}' 뉴스가 자동으로 업데이트되었습니다.`, "success");
                }
                updateCountdownDisplay();
            }, 1000);
        }

        function updateCountdownDisplay() {
            const mins = Math.floor(countdownSeconds / 60);
            const secs = countdownSeconds % 60;
            const formatted = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')} 남음`;
            document.getElementById('countdownDisplay').textContent = formatted;
        }

        // Custom Toast Notification System
        function showToast(message, type = "info") {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            toast.className = `pointer-events-auto flex items-center gap-3 px-4 py-3 rounded-xl shadow-2xl border text-xs font-medium transform transition-all duration-300 translate-y-2 opacity-0 bg-slate-900 text-slate-100 ${
                type === 'success' ? 'border-emerald-500/60' : type === 'warning' ? 'border-amber-500/60' : 'border-gold-500/60'
            }`;

            const iconClass = type === 'success' ? 'fa-circle-check text-emerald-400' : type === 'warning' ? 'fa-triangle-exclamation text-amber-400' : 'fa-circle-info text-gold-400';

            toast.innerHTML = `
                <i class="fa-solid ${iconClass} text-sm"></i>
                <span>${message}</span>
            `;

            container.appendChild(toast);

            // Animate In
            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            // Auto Remove
            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => {
                    toast.remove();
                }, 300);
            }, 3000);
        }

        // Utility Date formatters
        function formatDateStr(d) {
            const year = d.getFullYear();
            const month = String(d.getMonth() + 1).padStart(2, '0');
            const day = String(d.getDate()).padStart(2, '0');
            return `${year}-${month}-${day}`;
        }

        function formatTimeStr(d) {
            const hours = String(d.getHours()).padStart(2, '0');
            const minutes = String(d.getMinutes()).padStart(2, '0');
            return `${hours}:${minutes}`;
        }
